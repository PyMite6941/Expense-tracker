# ── Licence verification + rate limiting ─────────────────────────────────────
# PyJWT, NOT python-jose. python-jose 3.3.0 has three advisories on this exact
# path (PYSEC-2024-232 algorithm confusion, PYSEC-2024-233 JWE bomb,
# PYSEC-2025-185 JWE decrypt DoS with no fix) and depends on `ecdsa`, which has
# an unfixed Minerva timing attack (PYSEC-2026-1325). PyJWT needs only
# `cryptography`. HS256 is wire-identical, so licences already issued still verify.
PyJWT>=2.10,<3                          # verified 2.12.1
# Pinned above the chain-verification and OpenSSL advisories in 48.x.
cryptography>=50.0
slowapi>=0.1.9,<1.0.0

# ── Storage (licence IP-activation cap) ──────────────────────────────────────

# every one of them is optional at runtime (lazy imports, 503 or an LLM fallback).

requests>=2.32.0,<3.0.0
